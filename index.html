processing/silver/bronze_to_iceberg.pyPython · PySpark
# Bronze → Silver: Apache Iceberg with schema evolution & dedup
from pyspark.sql import SparkSession
import pyspark.sql.functions as F
from pyspark.sql.window import Window

def bronze_to_silver_iceberg(spark, bronze_path, silver_table, tenant_id, batch_date):
    df = (spark.read.format("delta").load(bronze_path)
          .filter((F.col("_batch_date") == batch_date) & (F.col("tenant_id") == tenant_id)))
    w = Window.partitionBy("event_id").orderBy(F.col("kafka_ts").desc())
    deduped = df.withColumn("_rn", F.row_number().over(w)).filter(F.col("_rn") == 1).drop("_rn")
    silver_df = deduped.select(
        "event_id", "tenant_id", "user_id", "event_type", "amount",
        F.current_timestamp().alias("_silver_ts"), F.lit(batch_date).alias("_batch_date"))
    (silver_df.writeTo(silver_table).option("mergeSchema", "true")
     .partitionedBy("tenant_id", "_batch_date").createOrReplace())
dq/quality_engine.pyPython · PySpark
# 9-rule DQ Engine — incidents −60%, confidence 82%→97% at Kenko AI
from dataclasses import dataclass
from enum import Enum
import pyspark.sql.functions as F

class RuleType(str, Enum):
    NOT_NULL="not_null"; UNIQUE="unique"; RANGE="range"; REGEX="regex"
    ACCEPTED_VALUES="accepted_values"; REFERENTIAL="referential_integrity"
    FRESHNESS="freshness"; SCHEMA="schema_check"; SEMANTIC="semantic"

@dataclass
class DQResult:
    rule_name: str; passed: bool; total_rows: int; failed_rows: int; failure_rate: float

class QualityEngine:
    def __init__(self, rules, alert_threshold=0.05):
        self.rules=rules; self.alert_threshold=alert_threshold

    def run(self, df, quarantine_path=None):
        total=df.count(); bad_mask=F.lit(False); results=[]
        for rule in self.rules:
            fail_cond=self._build_condition(rule)
            failed=df.filter(fail_cond).count(); rate=failed/total if total else 0.0
            results.append(DQResult(rule.name, rate<=self.alert_threshold, total, failed, round(rate,4)))
            bad_mask=bad_mask|fail_cond
        clean_df=df.filter(~bad_mask)
        if quarantine_path: df.filter(bad_mask).write.format("delta").mode("append").save(quarantine_path)
        return clean_df, results
ingestion/streaming/dms_cdc_to_bronze.pyPython · Structured Streaming
# AWS DMS CDC → Bronze Iceberg (real-time at Kenko AI)
import pyspark.sql.functions as F
from pyspark.sql.types import StructType, StringType, TimestampType

CDC_SCHEMA = (StructType().add("Op",StringType()).add("tenant_id",StringType())
              .add("member_id",StringType()).add("status",StringType()).add("updated_at",TimestampType()))

def run_cdc_stream(spark, s3_path, bronze_table):
    raw = spark.readStream.format("csv").schema(CDC_SCHEMA).option("header","true").load(s3_path)
    bronze = raw.filter(F.col("Op").isin(["I","U"])).select(
        "tenant_id","member_id","status",
        F.col("updated_at").alias("source_ts"), F.current_timestamp().alias("ingest_ts"),
        F.to_date("updated_at").alias("_date"))
    def write_batch(batch_df, _):
        batch_df.writeTo(bronze_table).option("mergeSchema","true").append()
    (bronze.writeStream.foreachBatch(write_batch)
     .option("checkpointLocation",f"s3://ckpts/{bronze_table}")
     .trigger(processingTime="60 seconds").start().awaitTermination())

processing/gold/revenue_mart.pyPython · PySpark · SQL
# Gold: MoM Revenue Mart — replaced GoodData, licensing −35%
import pyspark.sql.functions as F
from pyspark.sql.window import Window
from delta.tables import DeltaTable

def build_revenue_mart(spark, silver_path, gold_path, batch_date):
    silver = spark.read.format("delta").load(silver_path).filter(F.col("_batch_date")==batch_date)
    w = Window.partitionBy("tenant_id").orderBy("month")
    revenue = (silver.groupBy("tenant_id","studio_id",F.date_trunc("month","payment_date").alias("month"))
        .agg(F.sum("amount").alias("total_revenue"),F.countDistinct("member_id").alias("paying_members"))
        .withColumn("prev_revenue",F.lag("total_revenue").over(w))
        .withColumn("mom_growth_pct",F.round((F.col("total_revenue")-F.col("prev_revenue"))/F.nullif(F.col("prev_revenue"),F.lit(0))*100,2))
        .withColumn("_gold_ts",F.current_timestamp()))
    if DeltaTable.isDeltaTable(spark,gold_path):
        DeltaTable.forPath(spark,gold_path).alias("t").merge(revenue.alias("s"),
            "t.tenant_id=s.tenant_id AND t.studio_id=s.studio_id AND t.month=s.month"
        ).whenMatchedUpdateAll().whenNotMatchedInsertAll().execute()
    else:
        revenue.write.format("delta").partitionBy("tenant_id").save(gold_path)
