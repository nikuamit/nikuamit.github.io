dq/quality_engine.py Python · PySpark 
"""
Rule-based Data Quality engine.
Bad rows quarantined to Delta; summary metrics emitted per run.
"""
from pyspark.sql import DataFrame
import pyspark.sql.functions as F
from dq.models import DQRule, DQResult, RuleType

class QualityEngine:
    def __init__(self, rules, fail_on_error=False, alert_threshold=0.05):
        self.rules = rules
        self.fail_on_error = fail_on_error
        self.alert_threshold = alert_threshold

    def run(self, df: DataFrame, quarantine_path=None):
        total   = df.count()
        results = []
        bad_mask = F.lit(False)

        for rule in self.rules:
            fail_cond = self._build_fail_condition(rule, df)
            if fail_cond is None: continue

            failed = df.filter(fail_cond).count()
            rate   = failed / total if total else 0.0
            passed = rate <= self.alert_threshold

            results.append(DQResult(
                rule_name=rule.name, passed=passed,
                total_rows=total, failed_rows=failed, failure_rate=round(rate, 4)
            ))

            if not passed:
                bad_mask = bad_mask | fail_cond
                if self.fail_on_error:
                    raise ValueError(f"DQ rule '{rule.name}' failed: {rate:.2%} bad rows")

        clean_df     = df.filter(~bad_mask)
        quarantine_df = df.filter(bad_mask)

        if quarantine_path and quarantine_df.count() > 0:
            (quarantine_df
             .withColumn("_dq_ts", F.current_timestamp())
             .write.format("delta").mode("append").save(quarantine_path))

        return clean_df, results

What this does: Runs configurable rules (not_null, range, regex, accepted_values, unique) against any PySpark DataFrame. Bad rows are written to a Delta quarantine path with a timestamp for investigation. The engine never silently drops data — everything is accounted for. Used in the Bronze→Silver processing step across all entities.

ingestion/streaming/kafka_to_bronze.py Python · PySpark Structured Streaming 
"""
Reads raw events from Kafka → lands in Bronze Delta layer.
Uses foreachBatch for exactly-once Delta writes + checkpoint resume.
"""
def run(topic=None):
    cfg          = load_config()
    topic        = topic or cfg["kafka"]["topics"]["events"]
    bronze_path  = cfg["storage"]["base_path"] + f"/bronze/{topic}"
    checkpoint   = cfg["storage"]["base_path"] + f"/checkpoints/{topic}"

    spark = get_spark()

    raw_stream = (
        spark.readStream.format("kafka")
        .option("kafka.bootstrap.servers", cfg["kafka"]["bootstrap_servers"])
        .option("subscribe", topic)
        .option("startingOffsets", "earliest")
        .load()
    )

    bronze_stream = raw_stream.select(
        F.col("key").cast("string").alias("event_key"),
        F.col("value").cast("string").alias("event_payload"),
        F.col("topic"), F.col("partition"), F.col("offset"),
        F.col("timestamp").alias("kafka_ts"),
        F.current_timestamp().alias("ingest_ts"),
        F.to_date(F.col("timestamp")).alias("event_date"),
    )

    def write_bronze_batch(batch_df, batch_id):
        write_delta(batch_df, bronze_path, mode="append",
                    partition_by=["event_date"])

    (bronze_stream.writeStream
     .foreachBatch(write_bronze_batch)
     .option("checkpointLocation", checkpoint)
     .trigger(processingTime="30 seconds")
     .start()
     .awaitTermination())

What this does: Structured Streaming job that reads from Kafka and lands raw events into the Bronze Delta layer every 30 seconds. No transforms in Bronze — raw bytes + Kafka metadata only. foreachBatch gives exactly-once semantics. Checkpoint ensures resume on restart with no data loss or duplication.

utils/delta_writer.py Python · Delta Lake 
"""
Idempotent Delta Lake writer.
Supports append, overwrite, and merge-upsert in one interface.
"""
def write_delta(df, path, mode="append", partition_by=None, merge_keys=None):
    if mode == "upsert":
        if not merge_keys:
            raise ValueError("merge_keys required for upsert mode")
        return _upsert(df, path, list(merge_keys))

    writer = df.write.format("delta").mode(mode)
    if partition_by:
        writer = writer.partitionBy(*partition_by)
    writer.save(path)

def _upsert(df, path, merge_keys):
    spark     = df.sparkSession
    condition = " AND ".join(f"t.{k} = s.{k}" for k in merge_keys)

    if DeltaTable.isDeltaTable(spark, path):
        target = DeltaTable.forPath(spark, path)
        (target.alias("t")
               .merge(df.alias("s"), condition)
               .whenMatchedUpdateAll()
               .whenNotMatchedInsertAll()
               .execute())
    else:
        # First run — no target table yet, just write
        df.write.format("delta").save(path)

What this does: Single interface for all Delta writes — append, overwrite, or upsert. The upsert path uses Delta's MERGE to update existing rows and insert new ones atomically. If the target table doesn't exist yet (first run), it falls back to a plain write. This pattern makes every pipeline stage idempotent — safe to re-run on failure.

processing/gold/silver_to_gold.py → apply_scd2() Python · Delta Lake · SCD Type 2 
"""
SCD Type 2: closes old records, inserts new ones when tracked columns change.
Keeps full history — every version of every row is preserved.
"""
def apply_scd2(source_df, target_path, natural_keys, tracked_cols,
               effective_col="effective_from", expiry_col="effective_to",
               current_col="is_current"):

    high_date = "9999-12-31"
    incoming  = source_df.withColumn(
        "_row_hash", F.md5(F.concat_ws("|", *[F.col(c) for c in tracked_cols]))
    )

    if not DeltaTable.isDeltaTable(spark, target_path):
        # Initial load — all rows are current
        (incoming
         .withColumn(effective_col, F.current_date())
         .withColumn(expiry_col, F.lit(high_date).cast("date"))
         .withColumn(current_col, F.lit(True))
         .write.format("delta").save(target_path))
        return

    target  = DeltaTable.forPath(spark, target_path)
    join_c  = " AND ".join(f"t.{k} = s.{k}" for k in natural_keys)
    close_c = join_c + f" AND t.{current_col} = true AND s._row_hash != t._row_hash"

    # Step 1: close changed current records
    (target.alias("t")
           .merge(incoming.alias("s"), close_c)
           .whenMatchedUpdate(set={expiry_col: "current_date()",
                                   current_col: "false"})
           .execute())

    # Step 2: insert new / changed rows as current
    (incoming
     .withColumn(effective_col, F.current_date())
     .withColumn(expiry_col, F.lit(high_date).cast("date"))
     .withColumn(current_col, F.lit(True))
     .drop("_row_hash")
     .write.format("delta").mode("append").save(target_path))

What this does: Implements SCD Type 2 on Delta Lake — the dimensional modelling pattern that preserves full history. When a tracked column changes, the old record is closed (expiry date = today, is_current = false) and a new record is inserted (effective_from = today, expiry = 9999-12-31). MD5 hash comparison avoids unnecessary row closes. Used for dimension tables in the Gold layer.

orchestration/pipeline.py Python · Airflow-ready 
"""
Lightweight pipeline runner with retry + exponential backoff.
Each Stage wraps into an Airflow PythonOperator with no changes.
"""
class Stage:
    def __init__(self, name, fn, retries=1):
        self.name    = name
        self.fn      = fn
        self.retries = retries

    def execute(self, **kwargs):
        for attempt in range(1, self.retries + 2):
            try:
                t0 = time.time()
                self.fn(**kwargs)
                log.info("stage complete", extra={"stage": self.name,
                          "elapsed_s": round(time.time() - t0, 2)})
                return
            except Exception as exc:
                log.error("stage failed", extra={"attempt": attempt, "error": str(exc)})
                if attempt > self.retries: raise
                time.sleep(2 ** attempt)  # exponential backoff

def build_batch_pipeline(entity):
    pipeline = Pipeline(name=f"{entity}_batch")
    pipeline.add(Stage("bronze_to_silver",
                        lambda batch_date, **kw: to_silver(entity, batch_date),
                        retries=1))
    pipeline.add(Stage("silver_to_gold",
                        lambda batch_date, **kw: build_aggregate(
                            silver_entity=entity, gold_name=f"{entity}_daily",
                            group_cols=["_batch_date"],
                            agg_exprs={"record_count": "count(1)"},
                            batch_date=batch_date),
                        retries=1))
    return pipeline

What this does: A minimal pipeline orchestrator — Stage/Pipeline classes with retry logic and exponential backoff. Each Stage is a plain Python callable, so it drops into Airflow as a PythonOperator with zero refactoring. Structured JSON logs at every stage start/end/fail give full observability without a separate monitoring tool.
